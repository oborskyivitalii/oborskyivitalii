const fs = require('node:fs');

const { manifest } = require(root + '/tools/quality/artifact.cjs');
const reportPath = '/workspace/scratch/d8631dcc2123/bfcache-check.json';
const report = {
  scope: 'Actual full-document Back/Forward; persisted pageshow distinguishes real bfcache from reload. No synthetic lifecycle events.',
  source: cp.execFileSync('git', ['rev-parse','HEAD'], {cwd:root,encoding:'utf8'}).trim(),
  tree: cp.execFileSync('git', ['rev-parse','HEAD^{tree}'], {cwd:root,encoding:'utf8'}).trim(),
  dirty: !!cp.execFileSync('git', ['status','--porcelain'], {cwd:root,encoding:'utf8'}).trim(),
  publicDigest: manifest(process.env.SITE_PUBLIC_DIR || root + '/docs').artifactDigest,

  cases: [], errors: [],

async function active(page) {
  const goal = await page.evaluate(() => {
    const element = document.querySelector('[data-scroll-entry="waiting"]');
    return element ? element.getBoundingClientRect().top + scrollY - innerHeight * 0.78 : null;
  });
  assert.notEqual(goal, null);
  let y = await page.evaluate(() => scrollY);
  while (y < goal) {
    y = Math.min(goal, y + 160);
    await page.evaluate(top => scrollTo({top, behavior:'instant'}), y);
    await page.waitForTimeout(20);
  }
  await page.waitForFunction(() => SiteScrollEntry.diagnostics().active > 0, null, {polling:10,timeout:2000});
}
async function main() {
  const {server,url} = await start();
  try {
    for (const engine of ['chromium','webkit']) {
      const browser = await playwright[engine].launch({headless:true,
        ...(engine === 'chromium' ? {executablePath:'/tmp/chromium',args:['--no-sandbox'],ignoreDefaultArgs:['--disable-back-forward-cache']} : {executablePath:'/workspace/scratch/db6290e7c2fe/webkit-launch.sh'})
      });
      try {
        for (const width of [1440,390]) {
          const context = await browser.newContext({viewport:{width,height:900}});
          await context.addInitScript(() => {
            window.__entryHistoryEvents = [];
            window.__entryDocument = Math.random().toString(36).slice(2);
            window.addEventListener('pageshow', event => __entryHistoryEvents.push({kind:'pageshow',persisted:event.persisted}));
            window.addEventListener('pagehide', event => __entryHistoryEvents.push({kind:'pagehide',persisted:event.persisted}));
          });
          const page = await context.newPage();
          page.on('pageerror', e => report.errors.push(String(e)));
          await page.goto(url + '/writing.html');
          await page.waitForFunction(() => SiteScene?.motionState?.().ready);
          await active(page);
          const before = await page.evaluate(() => ({document:__entryDocument,y:scrollY,diagnostics:SiteScrollEntry.diagnostics()}));
          await page.goto(url + '/credits.html');
          await page.goBack({waitUntil:'domcontentloaded'});
          await page.waitForTimeout(120);
          const after = await page.evaluate(() => ({document:__entryDocument,y:scrollY,diagnostics:SiteScrollEntry.diagnostics(),events:__entryHistoryEvents,waiting:document.querySelectorAll('[data-scroll-entry]').length,rows:document.querySelectorAll('.publication').length}));
          const persisted = after.events.some(e=>e.kind==='pageshow'&&e.persisted);
          assert.ok(after.rows > 0, 'native archive remains readable');
          assert.equal(after.diagnostics.active, 0, 'back landing is settled');
          if (persisted) {
            assert.equal(after.document, before.document);
            assert.equal(after.waiting, 0, 'real bfcache has no replay');
            assert.equal(after.diagnostics.observing, false, 'settled bfcache has no observer');
          }
          report.cases.push({engine,browser:browser.version(),width,before,after,persisted,status:'pass',physicalDevice:false});
          fs.writeFileSync(reportPath, JSON.stringify(report,null,2)+'\n');
          console.log(engine,width,'persisted',persisted);
          await context.close();
        }
      } finally { await browser.close(); }
    }
    assert.deepEqual(report.errors, []);
    report.pass = true;
  } catch(error) { report.pass=false; report.failure=error.stack; throw error; }
  finally { fs.writeFileSync(reportPath,JSON.stringify(report,null,2)+'\n'); server.close(); }
}
main().catch(e=>{console.error(e.stack);process.exitCode=1});
