import json, subprocess
from collections import Counter
from pathlib import Path
x = json.loads(Path('review/issue-65/2026-10-09-overhead-input.json').read_text())
def category(path):
    for r in x['measurement']['ordered_rules']:
        match = path in r.get('exact', []) or any(path.startswith(p) for p in r.get('prefix', []))
        if match and (not r.get('suffix') or any(path.endswith(s) for s in r['suffix'])):
            return r['category']
    raise ValueError(path)
for pr in x['pull_requests']:
    m = pr['metadata']
    raw = subprocess.check_output(['git', 'diff', '--numstat', '-z', m['base_sha'], m['head_sha']])
    parts, i, totals = raw.split(b'\0'), 0, Counter()
    while i < len(parts) and parts[i]:
        a, d, path = parts[i].split(b'\t', 2)
        i += 1
        if not path:
            old, path = parts[i:i+2]  # Git rename record; destination owns classification.
            i += 2
        name, binary = path.decode(), a == b'-'
        c = category(name)
        totals[(c, 'files')] += 1
        totals[(c, 'binary_files')] += int(binary)
        if not binary:
            totals[(c, 'additions')] += int(a)
            totals[(c, 'deletions')] += int(d)
    print(m['number'], dict(sorted(totals.items())))
